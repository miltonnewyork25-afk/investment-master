import numpy as np
from scipy import stats

    [1.00, 0.45, -0.25, 0.30],  # AWS增长率

L = np.linalg.cholesky(correlation_matrix)

    # 生成独立标准正态分布
    independent_randoms = np.random.standard_normal(4)

    # 通过Cholesky分解引入相关性

    # 转换为具体分布

    valuation = dcf_calculation(aws_growth, ad_growth, wacc, terminal_g)
