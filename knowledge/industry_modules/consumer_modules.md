# 核心模块 M1-M10 (MECE, 任何消费品公司必须覆盖)

core_modules:

  M1_brand_identity:
    name: 品牌身份与定位
    definition: >
      公司是谁——品牌在消费者心智中的位置、情感连接强度、差异化来源。
      回答"为什么消费者选你而非替代品"。
    must_ask:
      - Q1: 消费者提到该品牌时的第一联想是什么？这个联想在过去5年是增强还是稀释？
      - Q2: 品牌强度(B轴)5维度评分——认知度/偏好度/忠诚度/差异化/情感度，哪项最弱？
      - Q3: 品牌弹性半径(R1/R2/R3/R4)各圈收入占比如何？核心圈外的延伸是增长引擎还是品牌稀释？
    kpi:
      - name: 品牌弹性半径评分
        formula: "BER = R1收入占比×1 + R2收入占比×3 + R3收入占比×5 (归一化1-10)"
        benchmark: ">7=平台级; 5-7=行业高端; 3-4=有限; <3=单品类锁定"
      - name: NPS (净推荐值)
        formula: "推荐者% - 贬损者%"
        benchmark: ">50=优秀; 30-50=良好; <30=需关注"
      - name: 品牌第一提及率
        formula: "未提示下品类联想中首先提及该品牌的消费者比例"
        benchmark: ">30%=品类领导; 15-30%=强势; <15%=弱势"
    consistency_check: >
      品牌弹性半径评分 ≤3 的公司，其收入增速不应长期超过核心品类增速的1.5倍
      (否则说明R2/R3收入被低估或品类定义有误)。
    kill_switch:
      threshold: "NPS连续3季同比下降且绝对值跌破20"
      action: "触发品牌衰退深度诊断——拆解是产品问题/渠道问题/竞品替代/代际偏好迁移"

  M2_pricing_power:
    name: 定价权与价格弹性
    definition: >
      公司在不显著损失销量的前提下提价的能力。
      消费品最核心的单一变量——直接决定利润率和通胀保护能力。
    must_ask:
      - Q1: 过去5年有机收入增长中，价格/组合(price/mix)贡献占比vs销量(volume)贡献占比？
      - Q2: 最近一次显著提价后(>3%)，6个月内销量变化幅度？价格弹性系数是多少？
      - Q3: 定价权的来源是什么——品牌情感溢价/品类无替代/渠道控制/转换成本？该来源是否可持续？
    kpi:
      - name: 价格弹性系数
        formula: "%Δ销量 / %Δ价格"
        benchmark: "绝对值<1.0=弱弹性(定价权强); 1.0-1.5=中等; >1.5=高弹性(定价权弱)"
      - name: 定价贡献率
        formula: "price/mix驱动的有机增长 / 总有机增长"
        benchmark: "30-60%=健康均衡; >80%=纯靠提价(量缩风险); <20%=定价权未释放"
      - name: 毛利率vs行业均值差
        formula: "公司毛利率 - 子行业平均毛利率"
        benchmark: ">10pp=强溢价; 3-10pp=中等; <3pp=无显著溢价"
    consistency_check: >
      定价贡献率>80%且销量连续2季负增长 → 必须标记"量价背离风险"。
      若此时毛利率仍在扩张，需验证是否为渠道去库存(一次性)而非需求萎缩(结构性)。
    kill_switch:
      threshold: "价格弹性系数绝对值>2.0 (每涨1%丢>2%销量)"
      action: "判定为商品化产品——降低品牌溢价系数至1.0, 估值切换至成本领先者框架"

  M3_channel_ecosystem:
    name: 渠道生态与分销效率
    definition: >
      产品/服务如何到达消费者——渠道结构、渠道议价力、全渠道整合程度、渠道成本效率。
    must_ask:
      - Q1: 收入按渠道拆分(直营/特许/批发/电商/DTC)的比例？各渠道利润率差异？
      - Q2: 渠道控制权在谁手里？公司对终端定价和品牌呈现的控制力如何？
      - Q3: 渠道变革(电商渗透/DTC转型/特许化)对利润率结构的净影响方向和幅度？
    kpi:
      - name: 电商渗透率
        formula: "线上收入 / 总收入"
        benchmark: "因子类型而异: 零售>15%=达标; 品牌>25%=达标; 餐饮>30%=达标(含外卖)"
      - name: 每单履约成本
        formula: "履约总成本 / 订单数"
        benchmark: "趋势比绝对值重要——YoY应下降或持平"
      - name: 渠道利润率极差
        formula: "最高渠道OPM - 最低渠道OPM"
        benchmark: ">15pp=结构性渠道套利机会; <5pp=渠道已均质化"
    consistency_check: >
      渠道利润率加权平均 ≈ 公司整体OPM (±2pp以内)。
      若偏差>2pp，说明渠道拆分遗漏了公司级费用分摊或存在交叉补贴。
    kill_switch:
      threshold: "单一渠道收入占比>70%且该渠道利润率YoY下降>300bps"
      action: "渠道集中度风险预警——必须建模该渠道恶化情景对公司级OPM的影响"

  M4_consumer_behavior:
    name: 消费者行为与忠诚度
    definition: >
      消费者为什么买、多久买一次、愿意付多少——购买频率、复购率、LTV、消费决策心理。
    must_ask:
      - Q1: 核心客群画像(年龄/收入/地域)是什么？该客群在人口结构中是扩张还是收缩？
      - Q2: 复购率/续费率是多少？TOP20%客户贡献了多少收入(帕累托浓度)？
      - Q3: 消费者转换到替代品的实际成本(金钱+时间+习惯)有多高？是上升还是下降趋势？
    kpi:
      - name: 客户留存率/续费率
        formula: "(期末活跃客户 - 新增客户) / 期初活跃客户"
        benchmark: "会员制>90%=优秀; 订阅>85%=优秀; 零售用同店客流YoY代替"
      - name: 客户生命周期价值(LTV)
        formula: "客均年消费 × 毛利率 × 平均留存年数"
        benchmark: "LTV/CAC > 3:1=健康; 1-3:1=需优化; <1:1=不可持续"
      - name: TOP20%客户收入占比
        formula: "前20%客户贡献收入 / 总收入"
        benchmark: ">60%=高集中(忠诚但脆弱); 40-60%=健康; <40%=分散(低忠诚)"
    consistency_check: >
      LTV × 活跃客户数 应≈公司未来3-5年收入NPV(±20%)。
      偏差过大说明LTV假设(留存/消费/毛利)与公司级财务不一致。
    kill_switch:
      threshold: "会员续费率/核心客户留存率跌破80%"
      action: "客户流失危机——立即拆分流失原因(价格/体验/竞品/需求变化), 下调增长假设至行业底线"

  M5_operational_efficiency:
    name: 运营效率与规模经济
    definition: >
      公司将收入转化为利润的效率——成本结构、规模效应、供应链管理、运营杠杆。
    must_ask:
      - Q1: 固定成本vs可变成本比例？收入每增长1%，OPM扩张多少bps(运营杠杆系数)？
      - Q2: 关键运营效率指标(坪效/人效/库存周转)vs同行处于什么分位？趋势方向？
      - Q3: 供应链是否存在结构性优势(自有物流/独家供应商/垂直整合)？该优势的可复制性？
    kpi:
      - name: SG&A率
        formula: "SG&A / Revenue"
        benchmark: "因子类型而异: 零售<22%; 品牌<35%; 餐饮<25% (趋势比绝对值重要)"
      - name: 库存周转率
        formula: "COGS / 平均库存"
        benchmark: "会员仓储>11x; 大众零售>8x; 品牌>4x; 餐饮>50x"
      - name: ROIC
        formula: "NOPAT / 平均投入资本"
        benchmark: ">20%=优秀; 12-20%=良好; <12%=平庸(消费品标准)"
    consistency_check: >
      库存增速 - 销售增速 ≤ 2pp。若库存增速持续超过销售增速>2pp连续2季,
      说明渠道堵塞或需求减弱, 需检查是否有存货减值风险。
    kill_switch:
      threshold: "ROIC连续3年低于WACC"
      action: "价值毁灭确认——检查是否为投资期(可接受)还是结构性低回报(需下调估值倍数)"

  M6_growth_engine:
    name: 增长引擎与天花板
    definition: >
      收入增长的来源拆分和可持续性——同店增长vs新店/新品/新市场/新品类/M&A的贡献。
    must_ask:
      - Q1: 过去5年收入CAGR的归因拆分(同店/新店/新品类/并购/汇率)？哪个引擎在衰减？
      - Q2: 总可寻址市场(TAM)有多大？当前渗透率？TAM是在扩张还是收缩？
      - Q3: 增长是否需要持续资本投入(开店/并购)还是轻资产复制(特许/数字化)？增量ROIC趋势？
    kpi:
      - name: 同店销售增长(comp)
        formula: "开业≥1年门店(或可比基础)收入YoY"
        benchmark: ">3%=健康有机增长; 0-3%=温和; <0%连续2季=需深挖原因"
      - name: 新单元经济回报
        formula: "新店/新市场首年回报率 = 首年EBITDA / 初始投资"
        benchmark: ">25%=优秀; 15-25%=良好; <15%=扩张质量存疑"
      - name: TAM渗透率
        formula: "公司收入 / 总可寻址市场"
        benchmark: "<10%=早期(增长空间大); 10-30%=中期; >30%=成熟(增长依赖份额抢夺)"
    consistency_check: >
      同店增长 + 新店贡献 + 并购贡献 + 汇率影响 ≈ 总收入增长率(±1pp)。
      若对不上，说明增长归因拆分遗漏了某个驱动因素。
    kill_switch:
      threshold: "同店增长连续4季为负 且 新单元经济回报<15%"
      action: "双引擎熄火——有机增长和扩张增长同时失效, 重新评估整体增长假设, 估值切换至低增长/现金牛框架"

  M7_competitive_moat:
    name: 竞争护城河与战略取舍
    definition: >
      竞争优势的来源、深度、可持续性——以及公司为了护城河放弃了什么(战略取舍)。
    must_ask:
      - Q1: 护城河的主要类型是什么(品牌/规模/网络/转换/成本/特许)?该类型的5年衰减概率？
      - Q2: 公司为了核心战略"放弃了什么"(战略放弃清单)?至少识别5项,其中几项竞争者不可兼得？
      - Q3: 稳健比率(RR)是多少？护城河是"价值让渡型"(RR>3:1, 难匹配)还是"品牌依赖型"(RR<1:1, 可替代)？
    kpi:
      - name: 稳健比率(RR)
        formula: "(客户节省/客户 + 员工超额/客户) / (税前利润/客户)"
        benchmark: ">5:1=几乎不可挑战; 3-5:1=强; 1-3:1=中等; <1:1=品牌依赖型"
      - name: 战略放弃清单中"不可兼得"项占比
        formula: "竞争者无法两者兼得的放弃项 / 总放弃项"
        benchmark: ">50%=护城河核心坚实; 30-50%=中等; <30%=护城河多为可复制要素"
      - name: 市场份额稳定性
        formula: "过去5年市场份额波动标准差"
        benchmark: "σ<1pp=极稳定; 1-3pp=正常; >3pp=护城河受侵蚀"
    consistency_check: >
      稳健比率>3:1 的公司, 其W×C双轴评分的W(意愿)应≥4。
      若RR>3:1但W<3, 说明"高让渡"来自行业竞争被迫而非战略选择, 护城河判断需降级。
    kill_switch:
      threshold: "战略放弃清单中出现≥2项'正在收回'的放弃 (战略漂移)"
      action: "战略纪律警报——Phase 4红队必须挑战管理层战略一致性, 估值倍数降低10-15%反映执行风险"

  M8_management_culture:
    name: 管理层质量与文化制度化
    definition: >
      管理层的资本配置能力、战略执行力、文化制度化程度、CEO继任风险。
    must_ask:
      - Q1: 管理层资本配置的历史回报率(回购/并购/有机投资)?是否系统性创造or毁灭价值？
      - Q2: 文化可衡量性评分(CM1-CM5总分)？文化是制度化(可传承)还是领袖依赖(随CEO走)？
      - Q3: W×C双轴评分？管理层的"意愿"(对消费者/员工让利)是否与"能力"(运营效率)匹配？
    kpi:
      - name: 文化可衡量性总分(CMS)
        formula: "CM1规则清晰度 + CM2可验证性 + CM3继任韧性 + CM4优先级排序 + CM5一致性 (各1-5分)"
        benchmark: "20-25=制度化文化; 13-19=混合; 5-12=领袖依赖"
      - name: 资本配置回报率
        formula: "增量ROIC — 过去5年新增投入资本的NOPAT回报"
        benchmark: ">WACC+5pp=优秀; WACC±5pp=合格; <WACC=价值毁灭"
      - name: W×C意愿×能力评分
        formula: "W轴(1-5) × C轴(1-5)"
        benchmark: "W×C≥16=王者象限; 9-15=中间; <9=需额外风险折价"
    consistency_check: >
      CMS<12(领袖依赖)的公司, CEO任期内的业绩波动应>CMS≥20公司。
      若CMS<12但业绩极其稳定, 需重新审视CMS评分是否低估了制度化程度。
    kill_switch:
      threshold: "CMS<12 且 CEO宣布离任/被替换"
      action: "领袖断裂风险——强制新增CEO继任风险折价(估值-10~-20%), 直到新CEO证明执行力(≥4季)"

  M9_financial_resilience:
    name: 财务韧性与现金创造
    definition: >
      资产负债表健康度、现金流质量、债务结构、股东回报能力——公司穿越周期的财务弹药。
    must_ask:
      - Q1: 自由现金流转化率(FCF/Net Income)？FCF质量如何(运营vs一次性)?
      - Q2: 净债务/EBITDA？债务到期分布是否集中？利率环境变化对利息成本的敏感度？
      - Q3: 如果发生行业寒冬(收入下降20%), 公司能维持运营多少个季度不需外部融资？
    kpi:
      - name: FCF转化率
        formula: "Free Cash Flow / Net Income"
        benchmark: ">90%=优秀; 70-90%=良好; <70%=质量存疑(查应计项目)"
      - name: 净债务/EBITDA
        formula: "(总债务 - 现金) / EBITDA"
        benchmark: "<1.5x=保守; 1.5-3.0x=适中; >3.0x=激进; 负权益公司用EBITDA覆盖率替代"
      - name: 股东总回报率
        formula: "(回购 + 股息) / 市值"
        benchmark: ">5%=高回报; 3-5%=正常; <2%=低回报(需用增长补偿)"
    consistency_check: >
      FCF转化率>100%持续多年 → 检查是否存在应付账款异常扩张(占用供应商资金)
      或资本支出被系统性延迟(维护性CapEx不足)。
    kill_switch:
      threshold: "净债务/EBITDA>4.0x 且 FCF转化率<70%"
      action: "财务压力预警——检查债务到期墙+评级展望, 估值模型必须纳入再融资风险情景"

  M10_valuation_anchor:
    name: 估值锚定与市场预期
    definition: >
      市场隐含的增长/利润率/回报假设是什么(逆向估值)——与M1-M9的基本面判断对比,
      判断估值是偏贵还是偏便宜。
    must_ask:
      - Q1: 逆向DCF反推的隐含增长率/OPM/ROIC分别是什么？这些假设在M1-M9分析中是否合理？
      - Q2: 当前估值倍数(EV/EBITDA, P/E)处于自身历史和同业对比的什么分位？
      - Q3: 市场定价中最大的"隐含赌注"是什么？该赌注的实现概率(基于M1-M9证据)？
    kpi:
      - name: 逆向隐含增长率
        formula: "使当前市值=DCF的收入CAGR假设"
        benchmark: "与行业增速比: 隐含>行业2x=乐观; 0.5-2x=中性; <0.5x=悲观"
      - name: EV/EBITDA vs 5年中位数
        formula: "当前EV/EBITDA / 5年滚动中位数"
        benchmark: ">1.2x=偏贵; 0.8-1.2x=合理; <0.8x=偏便宜"
      - name: 期望回报率
        formula: "(概率加权EV - 当前市值) / 当前市值"
        benchmark: ">+30%=深度关注; +10~+30%=关注; -10~+10%=中性; <-10%=审慎"
    consistency_check: >
      概率加权EV中各情景的增长/OPM假设必须与M6(增长)和M5(效率)中的分析一致。
      若乐观情景OPM=25%但M5分析得出OPM天花板=22%, 需修正估值或补充解释。
    kill_switch:
      threshold: "逆向隐含增长率>TAM增速 (市场定价超过市场本身增长)"
      action: "过度乐观定价——必须在报告中明确标注'市场隐含份额持续扩张', 评估该假设的现实性"

extension_modules:

  E1_franchise_economics:
    name: 特许经营/瓶装商经济学
    definition: >
      特许模式的双层利润结构、特许方与被特许方的利益一致性、再特许化进度。
    trigger: >
      公司收入中特许/授权/浓缩液占比>20%, 或商业模式包含特许-直营混合结构。
